## 상관서브쿼리

```sql
-- 상관 서브쿼리2
-- 한 번이라도 주문된 상품 조회하기
select product_id, name, price
from products
where product_id in (select -- 메인쿼리가 크고 서브쿼리가 작을 때 사용
						distinct product_id 
					from orders); 

select product_id, name, price
from products p
where exists( -- 메인쿼리가 작고 서브쿼리가 클 때 사용
	select 1
    from orders o
    where o.product_id = p.product_id
);
```

## SELECT 서브쿼리

```sql
-- select 서브쿼리
select 
	name,
    price,
    (select avg(price) from products) as avg_price -- 스칼라 서브쿼리만 가능
from products;

-- join이 너무 복잡하거나, 완전히 다른 테이블에서 간단한 정보 하나만 조회해 올 떄 코드를 명료하게 만들어줌
select
	p.product_id,
    p.name,
    p.price,
    (select count(*) 
		from orders o 
        where o.product_id = p.product_id) as order_count
from products p;

-- join으로 해결하는 방법
select
	p.product_id, 
    p.name, 
    p.price, 
    count(o.order_id) as order_count
from products p
left join orders o on p.product_id = o.product_id
group by p.product_id, p.name, p.price;
```
