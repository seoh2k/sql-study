## 단순 else문

```sql
-- 단순 else문
select 
	order_id, 
	user_id, 
    product_id, 
    quantity, 
    status,
    CASE status
		when 'PENDING' then '주문대기'
		when 'COMPLETED' then '결제완료'
		when 'SHIPPED' then '배송'
		when 'CANCELLED' then '주문취소'
        else '알 수 없음'
	end as status_korean
from orders;
```

## 검색 else문

```sql
-- 검색 case문
-- 가장 먼저 참이 되는 조건을 만나는 순간 실행을 멈춘다
-- 더 구체적인(범위가 좁은) 조건을 먼저 배치하는 것이 일반적
select 
	name,
    price,
    case 
		when price >= 100000 then '고가'
        when price >= 30000 then '중가'
        else '저가'
	end as price_label
from products;
```

## 사용 위치

```sql
-- CASE 문과 사용 위치
select 
	name,
    price,
    case 
		when price >= 100000 then '고가'
        when price >= 30000 then '중가'
        else '저가'
	end as price_label
from products
order by 
	case
		when price >= 100000 then 1
        when price >= 30000 then 2
        else 3
	end asc;
```

## 그룹핑

```sql
-- CASE 문 - 그룹핑
-- 1단계: CASE 문으로 각 고객 분류하기
select 
	name,
    birth_date,
    case 
		when year(birth_date) >= 1990 then '1990년대생'
		when year(birth_date) >= 1980 then '1980년대생'
		else '그 이전 출생'
	end as birth_decate
from users;

-- 2단계: GROUP BY 로 그룹화하고 COUNT 로 집계하기    
select 
	case 
		when year(birth_date) >= 1990 then '1990년대생'
		when year(birth_date) >= 1980 then '1980년대생'
		else '그 이전 출생'
	end as birth_decate,
    count(*) as customer_count
from users
group by 
	case 
		when year(birth_date) >= 1990 then '1990년대생'
		when year(birth_date) >= 1980 then '1980년대생'
		else '그 이전 출생'
	end;

-- 3단계: SELECT 절에서 AS 로 지정한 별칭( birth_decade )을 바로 사용
select 
	case 
		when year(birth_date) >= 1990 then '1990년대생'
		when year(birth_date) >= 1980 then '1980년대생'
		else '그 이전 출생'
	end as birth_decate,
    count(*) as customer_count
from users
group by birth_decate

```

## 조건부 집계1

```sql
-- CASE 문 - 조건부 집계1
-- UNION으로 합치는 방법
select 'Total' as category, count(*) as count from orders
union all
select status as category, count(*) as count from orders group by status;

-- 서브 쿼리를 사용하는 방법 -> 비효율적
select 
	(select count(*) from orders) as total_orders,
    (select count(*) from orders where status='COMPLETED') as compeleted_count,
    (select count(*) from orders where status='SHIPPED') as SHIPPED_count,
    (select count(*) from orders where status='PENDING') as PENDING_count;
```
