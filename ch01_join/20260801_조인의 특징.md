# 조인의 특징

생성일: 2026년 8월 1일 오전 6:40
날짜: 2026년 8월 1일
태그: SQL, 조인

```jsx
-- 자식 -> 부조 조인은 행 개수가 유지
-- FK -> PK 조인
select
	o.order_id,
    o.product_id,
    o.user_id as orders_user_id,
    u.user_id as users_user_id,
    u.name,
    u.email
from orders o
join users u on o.user_id = u.user_id
where o.user_id = 1;

-- 부모 -> 자식 조인은 행 개수 증가
-- PK -> FK 조인
select
	u.user_id as users_user_id,
    u.name,
    u.email,
	o.order_id,
    o.product_id,
    o.user_id as orders_user_id
from users u
join orders o on u.user_id = o.user_id
where o.user_id = 1;
```
